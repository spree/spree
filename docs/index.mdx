export const CopyBox = ({ label, command }) => {
  const [copied, setCopied] = useState(false);
  return (
    <div>
      <div className="font-mono text-xs uppercase tracking-[0.14em] text-[#999999] dark:text-zinc-500 mb-2">{label}</div>
      <button
        onClick={() => { navigator.clipboard.writeText(command); setCopied(true); setTimeout(() => setCopied(false), 2000); }}
        className="group/copy flex w-full items-center justify-between gap-4 rounded-xl border border-[#e2dfd8] dark:border-[#3f3e3c] bg-[#f7f5f2] dark:bg-[#272626] px-4 py-3.5 text-left cursor-pointer transition-colors hover:border-[#d5d5d5] dark:hover:border-[#565351]">

        <span className="shrink-0 text-[#999999] group-hover/copy:text-[#666666] dark:group-hover/copy:text-zinc-300">
          {copied
            ? <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="#22c55e" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><path d="M20 6 9 17l-5-5" /></svg>
            : <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><rect width="14" height="14" x="8" y="8" rx="2" ry="2" /><path d="M4 16c-1.1 0-2-.9-2-2V4c0-1.1.9-2 2-2h10c1.1 0 2 .9 2 2" /></svg>}
        </span>
      </button>
    </div>
  );
};
